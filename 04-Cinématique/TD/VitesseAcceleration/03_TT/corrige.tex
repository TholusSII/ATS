\begin{corrigebox}[Corrigé extrait de la banque]

\CorrigeQuestion{1}
Le point $C$ a un mouvement quelconque dans le plan $\left(A,\vi{0},\vj{0}\right)$.

\CorrigeQuestion{2}
On a
\[
\vect{AC}=\lambda(t)\vect{i_0}+\mu(t)\vect{j_0}.
\]
Ainsi, dans le repère $\repere{A}{i_0}{j_0}{k_0}$,
\[
\left\{
\begin{aligned}
x_C(t)&=\lambda(t),\\
y_C(t)&=\mu(t),\\
z_C(t)&=0.
\end{aligned}
\right.
\]

\CorrigeQuestion{3}
La norme de la vitesse vaut
\[
v=R\dot{\theta}(t).
\]
Par intégration, avec $\theta(0)=0$,
\[
\theta(t)=\frac{v}{R}t.
\]

\CorrigeQuestion{4}
La trajectoire circulaire s'écrit
\[
\vect{AC}=R\vect{e_r}
=R\cos\theta(t)\vi{0}+R\sin\theta(t)\vj{0}.
\]
Par identification,
\[
\boxed{
\lambda(t)=R\cos\left(\frac{v}{R}t\right),
\qquad
\mu(t)=R\sin\left(\frac{v}{R}t\right)
}.
\]

\CorrigeQuestion{5}
La durée d'un tour complet est
\[
T=\frac{2\pi R}{v}.
\]
On calcule ensuite les valeurs de $\lambda(t)$ et $\mu(t)$ sur
$t\in[0,T]$, puis on trace d'une part ces deux fonctions en fonction du
temps et d'autre part la courbe paramétrée
\[
\bigl(\lambda(t),\mu(t)\bigr).
\]
Cette dernière est un cercle de rayon $R$ centré en $A$.
