\begin{TLCarteMentale}{Lois entrée-sortie}
\TLBranche{Finalité}{Relier le paramètre d'entrée $q_e$ au paramètre de sortie $q_s$.}
\TLBranche{Modèle}{Bâti, solides, liaisons, repères et paramètres géométriques.}
\TLBranche{Entrée}{Angle, translation, vitesse ou déplacement imposé.}
\TLBranche{Sortie}{Grandeur de mouvement recherchée sur le solide utile.}
\TLBranche{Boucle géométrique}{Repérer une chaîne fermée et écrire une somme vectorielle nulle.}
\TLBranche{Projection}{Transformer l'équation vectorielle en relations scalaires exploitables.}
\TLBranche{Élimination}{Supprimer les variables intermédiaires pour obtenir $F(q_e,q_s)=0$.}
\TLBranche{Loi géométrique}{Exprimer $q_s=f(q_e)$ lorsque cela est possible.}
\TLBranche{Dérivation}{Dériver la loi géométrique pour relier $\dot q_e$ et $\dot q_s$.}
\TLBranche{Fermeture cinématique}{Composer les torseurs cinématiques dans une boucle fermée.}
\TLBranche{Rapport}{Calculer $r=\omega_s/\omega_e$ ou le rapport adapté aux grandeurs considérées.}
\TLBranche{Engrenages}{Utiliser les nombres de dents et le signe lié aux contacts extérieurs.}
\TLBranche{Puissance}{Vérifier la cohérence efforts-vitesses pour une transmission idéale ou réelle.}
\TLBranche{Singularités}{Repérer les configurations où la loi devient indéterminée ou très sensible.}
\TLBranche{Validation}{Contrôler signes, unités, limites, amplitudes et cohérence physique.}
\TLBranche{Résultat}{Loi de position, loi de vitesse, sens de mouvement et domaine d'utilisation.}
\end{TLCarteMentale}
