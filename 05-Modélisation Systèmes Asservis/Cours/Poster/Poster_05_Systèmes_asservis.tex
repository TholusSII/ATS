\begin{minipage}[t]{0.59\textwidth}
\begin{SApanel}[1. Définition]{SAblue2}
\justifying
Un \textbf{système asservi} élabore automatiquement son action à partir d'une
\textbf{consigne} et d'une \textbf{mesure} de la sortie. Il cherche à réduire
leur écart tout en restant performant malgré les perturbations et les
incertitudes du procédé.
\end{SApanel}
\end{minipage}\hfill
\begin{minipage}[t]{0.38\textwidth}
\begin{SAcard}[height=43mm,halign=center,valign=center]{SAblue2}
{\color{SAblue}\Large\bfseries Relation fondamentale}\par\vspace{3mm}
{\fontsize{21}{24}\selectfont\bfseries\(e(t)=r(t)-m(t)\)}\par\vspace{2mm}
{\color{SAgray}erreur = consigne $-$ mesure}
\end{SAcard}
\end{minipage}

\begin{minipage}[t]{0.49\textwidth}
\begin{SApanel}[4. Modèles essentiels]{SApurple}
\small
\begin{tabular}{@{}p{.34\linewidth}p{.60\linewidth}@{}}
\textbf{Fonction de transfert} & $H(p)=\dfrac{S(p)}{E(p)}$\\[2mm]
\textbf{Boucle fermée} & $H_{BF}(p)=\dfrac{G(p)}{1+G(p)H(p)}$\\[2mm]
\textbf{Premier ordre} & $H(p)=\dfrac{K}{1+\tau p}$\\[2mm]
\textbf{Second ordre} & $H(p)=\dfrac{K\omega_0^2}{p^2+2\xi\omega_0p+\omega_0^2}$
\end{tabular}
\end{SApanel}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\textwidth}
\begin{SApanel}[5. Démarche d'étude]{SAorange}
\small
\begin{enumerate}
  \item Identifier consigne, sortie, perturbations et capteur.
  \item Construire le schéma-bloc et déterminer la FTBO.
  \item Calculer la FTBF et repérer pôles et zéros.
  \item Étudier les réponses temporelle et fréquentielle.
  \item Conclure sur stabilité, précision, rapidité et robustesse.
\end{enumerate}
\end{SApanel}
\end{minipage}
