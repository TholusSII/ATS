\vspace{2mm}
\begin{minipage}[t]{0.61\linewidth}
\begin{TLPosterSavoirs}
\begin{itemize}
  \item distinguer source de tension et source de courant ; connaître les associations interdites et le rôle de $L$ et $C$ pendant les commutations
  \item connaître l'interrupteur idéal : passant $u_K=0$, bloqué $i_K=0$ ; diode spontanée, MOSFET/IGBT commandés, transistor + diode antiparallèle réversible en courant
  \item connaître $\alpha=t_{\rm on}/T$, valeur moyenne, valeur efficace et la propriété $\langle u_L\rangle=0$ en régime périodique établi
  \item reconnaître Buck, Boost, hacheur 2 quadrants, pont 4 quadrants, pont de diodes PD2 et structure PFC à Boost
  \item connaître $P=\langle ui\rangle$, $S=UI$, $\lambda=P/S$ et, en régime déformé, $S^2=P^2+Q^2+D^2$
\end{itemize}
\end{TLPosterSavoirs}
\end{minipage}\hfill
\begin{minipage}[t]{0.36\linewidth}
\TLPosterKey{Réflexe n°1}{Identifier la nature des sources et les réversibilités avant d'étudier les interrupteurs.}

\vfill
\begin{minipage}[t]{0.49\linewidth}
\begin{TLPosterBloc}[Choisir la structure]{TLPosterBlue}
\textbf{DC $\rightarrow$ DC :} Buck si $U_s<U_e$, Boost si $U_s>U_e$.\\
\textbf{Réversibilité :} 2 quadrants si une grandeur change de signe ; pont 4 quadrants si tension et courant doivent être réversibles.\\
\textbf{AC $\rightarrow$ DC :} pont de diodes pour redressement non commandé ; PFC si l'on veut un courant réseau quasi sinusoïdal et en phase.
\end{TLPosterBloc}
\begin{TLPosterBloc}[Méthode hacheur DC--DC]{TLPosterGreen}
Pour chaque intervalle : 1) indiquer passants/bloqués ; 2) remplacer les interrupteurs par court-circuit/circuit ouvert ; 3) redessiner le schéma équivalent ; 4) écrire $u_L=L\,di/dt$ ; 5) déduire les pentes et tracer les chronogrammes ; 6) imposer $\langle u_L\rangle=0$ pour relier entrée, sortie et $\alpha$ ; 7) calculer l'ondulation.
\end{TLPosterBloc}
\end{minipage}\hfill
\begin{minipage}[t]{0.49\linewidth}
\begin{TLPosterBloc}[Méthode redressement / puissance]{TLPosterPurple}
\textbf{PD2 :} repérer la diagonale conductrice sur chaque alternance, écrire $u_s(t)=|u_e(t)|$, puis calculer moyenne/efficace et contraintes des diodes. Pour une charge inductive, respecter la continuité du courant.\\
\textbf{Puissances :} partir de $p(t)=u(t)i(t)$ ; utiliser $P=UI\cos\varphi$ seulement si tension et courant sont sinusoïdaux.
\end{TLPosterBloc}
\begin{TLPosterBloc}[PFC, pertes et dimensionnement]{TLPosterTeal}
\textbf{PFC :} pont de diodes $\rightarrow$ Boost commandé $\rightarrow$ bus continu ; imposer $i_{\rm ref}(t)\propto|u_e(t)|$.\\
\textbf{Composants :} vérifier tension bloquée max, courant max/efficace, pertes de conduction et $P_{\rm com}=f_s(E_{\rm on}+E_{\rm off})$, puis température de jonction. Une fréquence plus élevée réduit les passifs mais augmente les pertes de commutation.
\end{TLPosterBloc}
\end{minipage}
\vfill
\begin{TLPosterMethode}
\begin{enumerate}
  \item \textbf{Identifier le cas :} DC/DC, AC/DC, conversion indirecte ; nature des sources, charge et réversibilités.
  \item \textbf{Si hacheur :} séquencer une période, établir les schémas équivalents, tracer $u$ et $i$, puis utiliser $\langle u_L\rangle=0$.
  \item \textbf{Si redresseur :} déterminer les diodes conductrices à chaque alternance, construire $u_s(t)$ et tenir compte de la continuité du courant si la charge est inductive.
  \item \textbf{Si étude de puissance :} calculer d'abord $P=\langle ui\rangle$ ; n'utiliser $P=UI\cos\varphi$ que dans le cas sinusoïdal.
  \item \textbf{Si qualité réseau :} calculer $\lambda=P/(UI)$, identifier harmoniques/distorsion et vérifier l'intérêt d'un PFC.
  \item \textbf{Toujours finir par :} ondulations, contraintes électriques, pertes, rendement, température et cohérence des signes.
\end{enumerate}
\end{TLPosterMethode}
