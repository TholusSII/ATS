% La conversion Word contient parfois plusieurs images consécutives dimensionnées
% chacune à \linewidth. On conserve les tailles explicites, on limite les images
% sans taille et on autorise une coupure entre deux illustrations.
\NewCommandCopy\TLLegacyIncludeGraphics\includegraphics
\RenewDocumentCommand{\includegraphics}{s O{} m}{%
  \ifstrempty{#2}{%
    \IfBooleanTF{#1}
      {\TLLegacyIncludeGraphics*[width=\linewidth,height=.82\textheight,keepaspectratio]{#3}}
      {\TLLegacyIncludeGraphics[width=\linewidth,height=.82\textheight,keepaspectratio]{#3}}%
  }{%
    \IfBooleanTF{#1}
      {\TLLegacyIncludeGraphics*[#2]{#3}}
      {\TLLegacyIncludeGraphics[#2]{#3}}%
  }%
  \ifinner\hspace{0pt}\allowbreak\else\par\noindent\fi
}
