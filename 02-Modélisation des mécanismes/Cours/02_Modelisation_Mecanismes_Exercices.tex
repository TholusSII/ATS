\NewCommandCopy\TLLegacyIncludeGraphics\includegraphics

\NewDocumentCommand{\TLLegacyRawGraphic}{s O{} m}{%
  \ifstrempty{#2}{%
    \IfBooleanTF{#1}
      {\TLLegacyIncludeGraphics*[width=\linewidth,height=.82\textheight,keepaspectratio]{#3}}
      {\TLLegacyIncludeGraphics[width=\linewidth,height=.82\textheight,keepaspectratio]{#3}}%
  }{%
    \IfBooleanTF{#1}
      {\TLLegacyIncludeGraphics*[#2]{#3}}
      {\TLLegacyIncludeGraphics[#2]{#3}}%
  }%
}

\newcommand{\TLLegacyMissingGraphic}[1]{%
  \begin{tikzpicture}[baseline=(current bounding box.center)]
    \node[
      draw=TLCoverBlue!55,
      rounded corners=1mm,
      fill=TLCoverBlue!4,
      text=TLCoverBlue!80!black,
      align=center,
      font=\sffamily\scriptsize,
      inner sep=2mm,
      text width=24mm
    ] {Illustration vectorielle\\non convertie\\\ttfamily #1};
  \end{tikzpicture}%
}

\newcommand{\TLLegacyPlaceGraphic}[1]{%
  \ifinner
    \adjustbox{max width=.90\linewidth,max totalheight=.72\textheight}{#1}%
    \hspace{0pt}\allowbreak
  \else
    \par\noindent
    \makebox[\linewidth][c]{%
      \adjustbox{max width=.96\linewidth,max totalheight=.82\textheight}{#1}%
    }%
    \par\noindent
  \fi
}

\makeatletter
\RenewDocumentCommand{\includegraphics}{s O{} m}{%
  \begingroup
  \filename@parse{#3}%
  \edef\TLLegacyGraphicExtension{\filename@ext}%
  \edef\TLLegacyGraphicStem{\filename@area\filename@base}%
  \edef\TLLegacyGraphicBase{\filename@base}%

  % \ifstrequal ne développe pas ses arguments : c'était la cause du passage
  % résiduel des fichiers .emf vers graphicx. \ifdefstring compare bien le
  % contenu développé de la macro d'extension.
  \TLLegacyVectorGraphicfalse
  \ifdefstring{\TLLegacyGraphicExtension}{emf}{\TLLegacyVectorGraphictrue}{}
  \ifdefstring{\TLLegacyGraphicExtension}{EMF}{\TLLegacyVectorGraphictrue}{}
  \ifdefstring{\TLLegacyGraphicExtension}{wmf}{\TLLegacyVectorGraphictrue}{}
  \ifdefstring{\TLLegacyGraphicExtension}{WMF}{\TLLegacyVectorGraphictrue}{}

  \ifTLLegacyVectorGraphic
    \IfFileExists{\TLLegacyGraphicStem.png}{%
      \TLLegacyPlaceGraphic{\TLLegacyIncludeGraphics[width=.90\linewidth,height=.72\textheight,keepaspectratio]{\TLLegacyGraphicStem.png}}%
    }{%
      \IfFileExists{\TLLegacyGraphicStem.pdf}{%
        \TLLegacyPlaceGraphic{\TLLegacyIncludeGraphics[width=.90\linewidth,height=.72\textheight,keepaspectratio]{\TLLegacyGraphicStem.pdf}}%
      }{%
        \IfFileExists{\TLLegacyGraphicStem.jpeg}{%
          \TLLegacyPlaceGraphic{\TLLegacyIncludeGraphics[width=.90\linewidth,height=.72\textheight,keepaspectratio]{\TLLegacyGraphicStem.jpeg}}%
        }{%
          \IfFileExists{\TLLegacyGraphicStem.jpg}{%
            \TLLegacyPlaceGraphic{\TLLegacyIncludeGraphics[width=.90\linewidth,height=.72\textheight,keepaspectratio]{\TLLegacyGraphicStem.jpg}}%
          }{%
            \TLLegacyPlaceGraphic{\TLLegacyMissingGraphic{\filename@base}}%
          }%
        }%
      }%
    }%
  \else
    % Les images 158 et 159 étaient exportées chacune à \linewidth sur une
    % même ligne. Elles sont traitées explicitement pour éviter tout débordement.
    \ifdefstring{\TLLegacyGraphicBase}{image158}{%
      \TLLegacyIncludeGraphics[width=.46\linewidth,height=.32\textheight,keepaspectratio]{#3}\hfill
    }{%
      \ifdefstring{\TLLegacyGraphicBase}{image159}{%
        \TLLegacyIncludeGraphics[width=.46\linewidth,height=.32\textheight,keepaspectratio]{#3}\par\noindent
      }{%
        \IfBooleanTF{#1}
          {\TLLegacyPlaceGraphic{\TLLegacyRawGraphic*[#2]{#3}}}
          {\TLLegacyPlaceGraphic{\TLLegacyRawGraphic[#2]{#3}}}%
      }%
    }%
  \fi
  \endgroup
}
