\CorrigeQuestion{2}
En écrivant la fermeture géométrique, on a
\[
\vect{AB}+\vect{BI}+\vect{IC}+\vect{CA}=\vect{0}.
\]
Soit
\[
e\vi{1}+R\vj{0}+\mu\vi{0}-\lambda(t)\vj{0}=\vect{0}.
\]
La projection sur $\vj{0}$ donne
\[
e\sin\theta(t)+R-\lambda(t)=0,
\]
d'où
\[
\boxed{\lambda(t)=e\sin\theta(t)+R}.
\]

\CorrigeQuestion{3}
En dérivant :
\[
\boxed{\dot{\lambda}(t)=e\dot{\theta}(t)\cos\theta(t)}.
\]

\CorrigeQuestion{4}
Le débit instantané est
\[
\boxed{q(t)=S\dot{\lambda}(t)=eS\dot{\theta}(t)\cos\theta(t)}.
\]

\CorrigeQuestion{5}
Pour une vitesse angulaire constante $\omega=\SI{100}{rad.s^{-1}}$, on a
$\theta(t)=\omega t$, donc
\[
q(t)=eS\omega\cos(\omega t).
\]
Avec $S=\SI{1e-4}{m^2}$, les amplitudes sont
\[
q_{\max}=\SI{1e-4}{m^3.s^{-1}}
\quad\text{pour } e=\SI{10}{mm},
\]
et
\[
q_{\max}=\SI{2e-4}{m^3.s^{-1}}
\quad\text{pour } e=\SI{20}{mm}.
\]
Le rayon $R$ intervient uniquement dans la position moyenne
$\lambda(t)=R+e\sin(\omega t)$ ; il ne modifie pas l'amplitude du débit.
