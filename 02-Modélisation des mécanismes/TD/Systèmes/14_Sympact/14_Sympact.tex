\question{Exprimer $\varphi(t)$ en fonction de $\theta(t)$.}
\ifprof
La fermeture géométrique s'écrit
\[
\vect{AB}+\vect{BC}+\vect{CA}=\vect{0},
\]
soit
\[
\lambda(t)\vect{i_2}-R\vect{i_1}-H\vect{j_0}=\vect{0}.
\]
Dans le repère $\rep{0}$, on obtient
\[
\left\{
\begin{aligned}
\lambda(t)\cos\varphi(t)&=R\cos\theta(t),\\
\lambda(t)\sin\varphi(t)&=R\sin\theta(t)+H.
\end{aligned}
\right.
\]
Par conséquent,
\[
\tan\varphi(t)=
\frac{R\sin\theta(t)+H}{R\cos\theta(t)}.
\]
On peut donc écrire, en tenant compte du quadrant,
\[
\varphi(t)=\operatorname{atan2}
\bigl(R\sin\theta(t)+H,\,R\cos\theta(t)\bigr).
\]
\fi

\question{Exprimer $\dot{\varphi}(t)$ en fonction de $\dot{\theta}(t)$.}
\ifprof
En dérivant la relation précédente, on obtient
\[
\boxed{
\dot{\varphi}(t)=
\frac{R\dot{\theta}(t)\left(R+H\sin\theta(t)\right)}
{R^2+H^2+2RH\sin\theta(t)}
}.
\]
\fi

\question{En utilisant Python, tracer $\dot{\varphi}(t)$ en fonction de $\dot{\theta}(t)$. On considérera que la fréquence de rotation de la pièce \textbf{1} est de 10 tours par minute.}
\ifprof
La vitesse angulaire d'entrée vaut
\[
\omega=10\times\frac{2\pi}{60}=\frac{\pi}{3}\ \mathrm{rad\,s^{-1}}.
\]
Pour le tracé, on pose $\theta(t)=\omega t$ et on évalue directement
\[
\dot{\varphi}(t)=
\frac{R\omega\left(R+H\sin(\omega t)\right)}
{R^2+H^2+2RH\sin(\omega t)}.
\]

\begin{solution}
\begin{enumerate}
\item Tracer le cycle des solides et identifier les pivots.
\item $\tan\varphi(t)=\dfrac{R\sin\theta(t)+H}{R\cos\theta(t)}$.
\item $\dot{\varphi}(t)=\dfrac{R\dot{\theta}(t)(R+H\sin\theta(t))}{R^2+H^2+2RH\sin\theta(t)}$.
\item Utiliser $\theta(t)=\omega t$ avec $\omega=\pi/3\ \mathrm{rad\,s^{-1}}$.
\end{enumerate}
\end{solution}}
