\begin{TLCarteMentale}{Analyse fonctionnelle}
\TLBranche{Besoin}{À qui le système rend-il service ? Sur quoi agit-il ? Dans quel but ?}
\TLBranche{Frontière du système}{Délimiter ce qui appartient au système et identifier les éléments extérieurs.}
\TLBranche{Parties prenantes}{Utilisateurs, opérateurs, mainteneurs, environnement et systèmes partenaires.}
\TLBranche{Contexte}{Représenter les interactions externes : matière, énergie, information et contraintes.}
\TLBranche{Cas d'utilisation}{Associer les acteurs aux services rendus par le système.}
\TLBranche{Fonctions de service}{Décrire les services principaux et complémentaires sans imposer de solution.}
\TLBranche{Contraintes}{Sécurité, réglementation, coût, ergonomie, énergie, environnement et maintenance.}
\TLBranche{Exigences}{Formaliser ce que le système doit satisfaire et assurer la traçabilité.}
\TLBranche{Critères et niveaux}{Rendre chaque fonction vérifiable par une grandeur mesurable et une valeur attendue.}
\TLBranche{Cahier des charges}{Regrouper fonctions, contraintes, critères, niveaux, priorités et flexibilités.}
\TLBranche{SysML}{Contexte, exigences, cas d'utilisation, définition de blocs et blocs internes.}
\TLBranche{Vérification}{Contrôler que la solution respecte les exigences spécifiées.}
\TLBranche{Validation}{S'assurer que le système répond réellement au besoin des parties prenantes.}
\end{TLCarteMentale}
