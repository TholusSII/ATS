% Les anciennes marginfigure de la banque sont transformées en blocs centrés
% non flottants. On définit directement \@captype au lieu d'appeler
% \captionsetup : avec certaines versions MiKTeX/caption/babel-french,
% \captionsetup déclenche une commande d'accent T1 à la fermeture de la minipage.
\newenvironment{marginfigure}{%
  \par\medskip\noindent
  \begin{minipage}{.82\linewidth}%
  \centering
  \def\@captype{figure}%
}{%
  \end{minipage}\par\medskip%
}

% Dans un recueil de plusieurs centaines de fichiers, les flottants historiques
% sont rendus non flottants tout en conservant \caption et \label.
\RenewDocumentEnvironment{figure}{O{}}{%
  \par\medskip\noindent\begin{center}%
  \def\@captype{figure}%
}{%
  \end{center}\par\medskip%
}
\RenewDocumentEnvironment{table}{O{}}{%
  \par\medskip\noindent\begin{center}%
  \def\@captype{table}%
}{%
  \end{center}\par\medskip%
}
\makeatother

% textuels cliquables, compacts et reproductibles.

  \href{#2}{\fbox{\strut\textbf{Accéder au questionnaire en ligne}}}%
}
\RenewDocumentCommand{\fancyqr}{O{} m}{%
  \href{#2}{\fbox{\strut\textbf{Accéder au questionnaire en ligne}}}%

% Dans un recueil, les figures et tableaux doivent rester avec leur exercice.
% Les environnements flottants historiques sont donc rendus non flottants.
% Cela évite aussi « Not in outer par mode » dans les corrigebox et les
% avertissements de modification automatique du placement [h] vers [ht].
\RenewDocumentEnvironment{figure}{O{}}{%
  \par\medskip\noindent\begin{center}\captionsetup{type=figure}%
}{%
  \end{center}\par\medskip%
}
\RenewDocumentEnvironment{table}{O{}}{%
  \par\medskip\noindent\begin{center}\captionsetup{type=table}%
}{%
  \end{center}\par\medskip%
}

  \textit{\color{gray!75!black}#1}%

  \par\smallskip\noindent{\scriptsize\color{gray!70!black}#1}\par\smallskip

\newcommand{\InclureCorrige}[1]{%
  \ifcorriges
    \par
    \IfFileExists{#1}{\input{#1}}{%
      \begin{corrigebox}[Corrigé indisponible]
        \CorrigeACompleter{Le fichier de corrigé attendu \texttt{#1} est absent.}
      \end{corrigebox}%
    }%
  \fi
}

% -----------------------------------------------------------------------------
% Compatibilité des blocs de code avec MiKTeX et l'encodage T1.
%
% environment_v2 contient une ancienne table listings « literate » qui remplace
% les caractères accentués par des commandes d'accent LaTeX. Certaines polices
% T1 de MiKTeX refusent ces commandes dès l'ouverture de lstlisting. On réinitialise
% réellement cette option (et non seulement sa valeur interne temporaire), puis on
% utilise une police machine à écrire en OT1, y compris pour verbatim.
% -----------------------------------------------------------------------------
\makeatletter
\@ifpackageloaded{listings}{%
  \lstset{
    literate={},
    inputencoding=utf8/latin1,
    upquote=false,
    basicstyle=\fontencoding{OT1}\ttfamily\small\selectfont
  }%
  \def\verbatim@font{\normalfont\fontencoding{OT1}\ttfamily\selectfont}%
}{}
\makeatother
